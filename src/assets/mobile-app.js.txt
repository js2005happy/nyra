class NetError extends Error {}
async function api(path, opts){
  opts = opts || {};
  let res;
  try { res = await fetch(path, { ...opts, headers: { 'X-Nyra-Token': token(), 'Content-Type': 'application/json' } }); }
  catch (e) { setOnline(false); throw new NetError(t('m.off.cant')); }
  let data = null; try { data = await res.json(); } catch(e){}
  if (res.status === 403) { lostDevice(); throw new Error((data && data.error) || t('m.unauthorized')); }
  setOnline(true);
  if (!res.ok) throw new Error((data && data.error) || ('HTTP ' + res.status));
  return data;
}

// base64url 编解码（配对加密用；btoa/atob 只认标准 base64）。
function b64urlEncode(bytes){ let s = ''; for (let i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]); return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, ''); }
function b64urlDecode(s){ const bin = atob(s.replace(/-/g, '+').replace(/_/g, '/')); const out = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i); return out; }

// 加密配对（POST /api/pair/encrypted）：手机生成临时 ECDH 密钥，把公钥随配对码
// 发去，网关用它把设备 token 加密后发回。WebCrypto 只在安全上下文可用；
// 明文 HTTP 下 crypto.subtle 是 undefined，调用方回落明文 /api/pair。
async function pairEncrypted(code){
  const kp = await crypto.subtle.generateKey({ name: 'ECDH', namedCurve: 'P-256' }, false, ['deriveBits']);
  const raw = new Uint8Array(await crypto.subtle.exportKey('raw', kp.publicKey)); // 65 字节 0x04||x||y
  const res = await fetch('/api/pair/encrypted', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code, name: deviceName(), public_key: b64urlEncode(raw.slice(1)) }) });
  const data = await res.json().catch(() => ({}));
  if (!res.ok || !data.envelope) throw new Error(data.error || t('m.pair.failed'));
  const env = data.envelope;
  const epkBytes = b64urlDecode(env.epk);
  const epkRaw = new Uint8Array(65); epkRaw[0] = 4; epkRaw.set(epkBytes, 1);
  const epk = await crypto.subtle.importKey('raw', epkRaw, { name: 'ECDH', namedCurve: 'P-256' }, false, []);
  const secret = await crypto.subtle.deriveBits({ name: 'ECDH', public: epk }, kp.privateKey, 256);
  const baseKey = await crypto.subtle.importKey('raw', secret, 'HKDF', false, ['deriveKey']);
  const aesKey = await crypto.subtle.deriveKey(
    { name: 'HKDF', hash: 'SHA-256', salt: epkBytes, info: new TextEncoder().encode('nyra-pair-v1') },
    baseKey, { name: 'AES-GCM', length: 256 }, false, ['decrypt']);
  const ct = b64urlDecode(env.ct), tag = b64urlDecode(env.tag);
  const buf = new Uint8Array(ct.length + tag.length); buf.set(ct); buf.set(tag, ct.length);
  const pt = await crypto.subtle.decrypt({ name: 'AES-GCM', iv: b64urlDecode(env.iv), tagLength: 128 }, aesKey, buf);
  return new TextDecoder().decode(pt);
}

async function pair(){
  const code = $('pair-code').value.replace(/\D/g, '');
  $('pair-error').textContent = '';
  if (code.length !== 6) { $('pair-error').textContent = t('m.pair.need6'); return; }
  $('pair-btn').disabled = true;
  try {
    let token;
    if (window.crypto && window.crypto.subtle) {
      try { token = await pairEncrypted(code); }
      catch (e) { if (e instanceof TypeError) throw e; $('pair-error').textContent = e.message || t('m.pair.failed'); return; }
    } else {
      // 明文回落：当前连接不是安全上下文，WebCrypto 不可用。配对页会显示
      // 未加密提示（m.pair.plainNote），不假装这条通道是加密的。
      const res = await fetch('/api/pair', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code, name: deviceName() }) });
      const data = await res.json().catch(() => ({}));
      if (!res.ok || !data.token) { $('pair-error').textContent = data.error || t('m.pair.failed'); return; }
      token = data.token;
    }
    try { localStorage.setItem('nyra-device-token', token); } catch(e){}
    history.replaceState(null, '', '/m');
    $('pair-code').value = ''; syncCodeBoxes();
    $('pair-lost').classList.add('hidden'); $('pair-title').textContent = t('m.pair.title');
    toast(t('m.pair.connected'));
    await boot();
  } catch (e) { $('pair-error').textContent = t('m.pair.cantReach', { msg: e.message }); }
  finally { $('pair-btn').disabled = false; }
}

function relTime(iso){
  if (!iso) return '';
  const d = new Date(/Z|[+-]\d\d:?\d\d$/.test(iso) ? iso : iso + 'Z');
  const s = Math.max(0, (Date.now() - d.getTime()) / 1000);
  if (!isFinite(s) || s < 60) return t('m.time.now');
  if (s < 3600) return t('m.time.m', { n: Math.floor(s / 60) });
  if (s < 86400) return t('m.time.h', { n: Math.floor(s / 3600) });
  return t('m.time.d', { n: Math.floor(s / 86400) });
}

// 一行的状态：[丸子表情, 状态词 class, 状态词, 说明]
function threadState(s){
  const a = lastApprovals.find(x => x.session === s.id);
  if (a) return ['wait', 'sw', t('m.st.wait'), waitText(a)];
  if (s.status === 'running') return ['work', 'sr', t('m.st.running'), t('m.st.runningSub')];
  if (s.status === 'error') return ['idle', 'se', t('m.st.error'), t('m.st.errorSub')];
  const seen = seenMap()[s.id];
  const fresh = seen !== undefined && (s.count || 0) > seen;
  const o = s.outcome;
  if (o && o.steps) {
    const push = t(o.pushed ? 'm.sub.pushed' : 'm.sub.notPushed');
    const txt = o.edited && o.edited.length ? t(o.edited.length === 1 ? 'm.sub.doneEdit1' : 'm.sub.doneEdit', { n: o.edited.length, push }) : (o.commands ? '' : t('m.sub.doneRead'));
    if (txt) return ['done', 'sd', t('m.st.done'), txt];
  }
  if (fresh) return ['done', 'sd', t('m.st.review'), ''];
  return ['idle', '', '', t('m.st.idle', { n: s.count || 0 })];
}

function liveTurn(v){
  const out = [];
  if (v.status !== 'running' || !Array.isArray(v.events)) return out;
  let last = -1; v.events.forEach((e, i) => { if (e.type === 'user') last = i; });
  if (last < 0) return out;
  v.events.slice(last).forEach(e => {
    const d = e.data || {};
    if (e.type === 'user') out.push({ role: 'user', content: String(d.content || '') });
    else if (e.type === 'tool_start') out.push({ role: 'assistant', content: '', tool_calls: [{ function: { name: String(d.name || '?'), arguments: typeof d.arguments === 'string' ? d.arguments : JSON.stringify(d.arguments || {}) } }] });
    else if (e.type === 'assistant') out.push({ role: 'assistant', content: String(d.content || '') });
  });
  return out;
}

function ruleLabel(sg){
  if (!sg) return '';
  if (sg.kind === 'prefix' && Array.isArray(sg.prefix)) return t('m.rule.prefix', { prefix: sg.prefix.join(' ') });

function parseDiff(summary){
  const lines = String(summary || '').split('\n');
  let path = '', add = 0, del = 0; const body = [];
  lines.slice(1).forEach(l => {
    if (l.startsWith('+++ ')) { path = l.replace(/^\+\+\+ (b\/)?/, ''); return; }
    if (l.startsWith('--- ')) { if (!path) path = l.replace(/^--- (a\/)?/, ''); return; }
    if (l.startsWith('+')) { add++; body.push(['a', '+', l.slice(1)]); }
    else if (l.startsWith('-')) { del++; body.push(['d', '−', l.slice(1)]); }
    else if (l.trim()) body.push(['', '', l]);
  });
  return { path, add, del, body, head: lines[0] || '' };
}
function apprBits(a){

  return { isEdit, isShell, d, sentence };
}

function scheduleText(task){
  if (task.at) return t('m.auto.daily', { at: task.at });
  const s = Number(task.every_seconds || 0);
  if (s >= 3600 && s % 3600 === 0) return t('m.auto.everyHour', { n: s / 3600 });
  return t('m.auto.everyMin', { n: Math.round(s / 60) });
}

function chooseMode(m){
  if (lastInfo && m === lastInfo.mode) return;
  if (m === 'full') { fullStep = 1; renderModes(); return; }
  setMode(m);

async function setMode(m){
  try {
    const data = await api('/api/mode', { method: 'POST', body: JSON.stringify({ mode: m }) });
    fullStep = 0;
    if (data && data.pending) {
      // 电脑确认门：请求已发给电脑，模式还没变；轮询 /api/info 等电脑确认。
      fullPending = { id: data.request_id, expiresAt: Date.parse(data.expires_at || '') };
      startFullPendingPoll();
      renderModes();
      return;
    }
    clearFullPending();
    await loadInfo(); renderModes(); renderCPills();
    toast(t('m.set.switched', { mode: modeLabel(m) }));
  } catch (e) { if (e instanceof NetError) actionFailed(e, () => setMode(m)); else toast(t('m.set.switchFailed', { msg: e.message })); }
}
// 待确认请求轮询：电脑批了就切过去，过期就提示重来。
function startFullPendingPoll(){
  if (fullPendingTimer) clearInterval(fullPendingTimer);
  fullPendingTimer = setInterval(() => { checkFullPending(); }, 5000);
}
function clearFullPending(){
  fullPending = null;
  if (fullPendingTimer) { clearInterval(fullPendingTimer); fullPendingTimer = null; }
}
async function checkFullPending(){
  if (!fullPending) return;
  try { await loadInfo(); } catch (e) { return; }
  if (lastInfo && lastInfo.mode === 'full') {
    clearFullPending(); renderModes(); renderCPills();
    toast(t('m.set.switched', { mode: modeLabel('full') }));
    return;
  }
  if (Date.now() > fullPending.expiresAt) {
    clearFullPending(); renderModes();
    toast(t('m.full.pendingExpired'));
  }
}

let chiefState = null;
function chiefAv(){ const a = $('chief-av'); a.innerHTML = ''; a.appendChild(mascot('chief', chiefState && chiefState.loading ? 'think' : 'idle', 30, { label: t('m.chief.name') })); }
async function openChief(task){
  currentSession = null; currentGroup = null;
  chiefState = { task, proposal: null, selected: [], loading: true, error: '' };
  show('chief-view'); chiefAv(); renderChief();
  try {
    const p = await api('/api/chief/proposals', { method: 'POST', body: JSON.stringify({ task }) });
    chiefState.proposal = p; chiefState.selected = p.members.map(m => m.id);
  } catch (e) { chiefState.error = e.message; }
  chiefState.loading = false; chiefAv(); renderChief();
}
function chiefMsg(text, state){
  const gm = el('div', 'gm'); gm.appendChild(mascot('chief', state || 'idle', 28, { label: t('m.chief.name') }));
  const bd = el('div', 'bd'); bd.appendChild(el('div', 'nm', t('m.chief.name'))); bd.appendChild(el('div', 'tx', text)); gm.appendChild(bd); return gm;
}
function permTag(){ const p = el('span', 'perm'); p.appendChild(icon('shield')); p.appendChild(document.createTextNode(t('m.perm.readonly'))); return p; }
function candidateOf(id){ return chiefState.proposal.candidates.find(c => c.id === id); }
function renderChief(){
  const feed = $('chief-feed'); feed.innerHTML = '';
  if (!chiefState) return;
  feed.appendChild(el('div', 'u', chiefState.task));
  if (chiefState.loading) { feed.appendChild(chiefMsg(t('m.chief.thinking'), 'think')); return; }
  if (chiefState.error) { const m = chiefMsg(chiefState.error); m.querySelector('.tx').classList.add('err'); feed.appendChild(m); return; }
  const p = chiefState.proposal;
  feed.appendChild(chiefMsg(t('m.chief.proposeLead', { n: chiefState.selected.length })));
  const card = el('div', 'card'); card.style.marginLeft = '38px'; card.id = 'chief-plan';
  const ch = el('div', 'ch'); const b = el('b'); b.appendChild(icon('users', 's18')); b.appendChild(document.createTextNode(t('m.chief.planTitle', { title: p.title }))); ch.appendChild(b);
  ch.appendChild(el('small', '', p.team ? t('m.chief.planSubTeam', { n: chiefState.selected.length, team: p.team.name }) : t('m.chief.planSub', { n: chiefState.selected.length })));
  card.appendChild(ch);
  chiefState.selected.forEach(id => {
    const c = candidateOf(id); if (!c) return;
    const row = el('div', 'mem'); row.dataset.member = id;
    row.appendChild(mascot(c.mascot, 'idle', 36, { label: c.name }));
    const tx = el('div', 'tx'); const nm = el('b', '', c.name); nm.appendChild(permTag()); tx.appendChild(nm); tx.appendChild(el('small', '', c.role)); row.appendChild(tx);
    const rm = el('button', 'ib'); rm.style.color = 'var(--t3)'; rm.setAttribute('aria-label', t('m.chief.remove', { name: c.name })); rm.dataset.act = 'remove-member'; rm.appendChild(icon('minus', 's18'));
    rm.disabled = chiefState.selected.length <= p.min;
    rm.onclick = () => { chiefState.selected = chiefState.selected.filter(x => x !== id); renderChief(); };
    row.appendChild(rm); card.appendChild(row);
  });
  const add = el('div', 'mem'); add.style.minHeight = '52px';
  const ab = el('button', 'tap row'); ab.style.cssText = 'gap:10px;color:var(--brand-t);font-weight:600;font-size:var(--f-m);justify-content:flex-start'; ab.dataset.act = 'add-member';
  ab.appendChild(icon('plus', 's18')); ab.appendChild(document.createTextNode(t('m.chief.addMember'))); ab.onclick = adjustMembers; ab.disabled = chiefState.selected.length >= p.max;
  add.appendChild(ab); add.appendChild(el('span', 'sp')); const mx = el('span', '', t('m.chief.max', { n: p.max })); mx.style.cssText = 'font-size:var(--f-xs);color:var(--t3);padding-right:10px'; add.appendChild(mx);
  card.appendChild(add);
  const note = el('div', 'note', t('m.chief.readonlyNote')); note.style.padding = '0 16px 10px'; card.appendChild(note);
  const act = el('div', 'row'); act.style.cssText = 'padding:4px 12px 12px;gap:8px';
  const adj = el('button', 'btn sm', t('m.chief.adjust')); adj.style.flex = '1'; adj.onclick = adjustMembers;
  const ok = el('button', 'btn sm pri', t('m.chief.create')); ok.style.flex = '1.6'; ok.dataset.act = 'chief-create'; ok.onclick = createChief;
  act.appendChild(adj); act.appendChild(ok); card.appendChild(act);
  feed.appendChild(card);
}
function adjustMembers(){
  const p = chiefState.proposal; if (!p) return;
  const draft = chiefState.selected.slice();
  const draw = () => {
    const sh = openSheet('members');
    sh.appendChild(el('h2', '', t('m.chief.adjustTitle')));
    sh.appendChild(el('p', 'sub-t', t('m.chief.adjustBody', { min: p.min, max: p.max })));
    const card = el('div', 'card'); card.style.cssText = 'margin-top:14px;background:var(--inset)';
    p.candidates.forEach(c => {
      const on = draft.includes(c.id);
      const row = el('button', 'mem'); row.dataset.member = c.id; row.setAttribute('role', 'checkbox'); row.setAttribute('aria-checked', String(on));
      row.appendChild(mascot(c.mascot, 'idle', 34, { label: c.name }));
      const tx = el('div', 'tx'); tx.appendChild(el('b', '', c.name)); tx.appendChild(el('small', '', c.role)); row.appendChild(tx);
      const box = el('span', 'tap'); const k = el('span', 'chk' + (on ? ' on' : '')); k.appendChild(icon('check')); box.appendChild(k); row.appendChild(box);
      row.onclick = () => {
        if (on) { if (draft.length <= p.min) { toast(t('m.chief.atLeast', { n: p.min })); return; } draft.splice(draft.indexOf(c.id), 1); }
        else { if (draft.length >= p.max) { toast(t('m.chief.atMost', { n: p.max })); return; } draft.push(c.id); }
        draw();
      };
      card.appendChild(row);
    });
    sh.appendChild(card);
    const perm = el('div', 'lbl'); perm.style.marginTop = '16px'; perm.textContent = t('m.chief.permTitle'); sh.appendChild(perm);
    const seg = el('div', 'seg'); seg.style.cssText = 'margin:0;grid-template-columns:1fr'; const only = el('button', 'on'); only.appendChild(icon('shield', 's16')); only.appendChild(document.createTextNode(t('m.perm.readonly'))); seg.appendChild(only); sh.appendChild(seg);
    sh.appendChild(el('p', 'sub-t', t('m.chief.permBody')));
    const done = el('button', 'btn pri full', t('m.done')); done.style.cssText = 'height:52px;margin-top:16px'; done.dataset.act = 'members-done';
    done.onclick = () => { chiefState.selected = draft.slice(); closeSheet(); renderChief(); };
    sh.appendChild(done);
  };
  draw();
}
async function createChief(){
  if (!chiefState || !chiefState.proposal) return;
  const btn = document.querySelector('[data-act="chief-create"]'); if (btn) btn.disabled = true;
  try {
    const res = await api('/api/chief/groups', { method: 'POST', body: JSON.stringify({ task: chiefState.proposal.task, members: chiefState.selected, team: chiefState.proposal.team ? chiefState.proposal.team.id : null }) });
    chiefState = null;
    await openGroup(res.room.id);
  } catch (e) { if (btn) btn.disabled = false; actionFailed(e, createChief); }
}
async function chiefReply(){
  const i = $('chief-input'); const v = i.value.trim(); if (!v) return; i.value = ''; autoGrow(i);
  await openChief(chiefState && chiefState.task ? chiefState.task + '\n' + v : v);
}

// ---------------------------------------------------------------- 群聊 ---
let groupData = null, groupKey = '';
async function openGroup(roomId){
  currentSession = null; currentGroup = roomId; groupData = null; groupKey = '';
  $('group-feed').innerHTML = ''; $('group-title').textContent = ''; $('group-sub').textContent = ''; $('group-stack').innerHTML = '';
  show('group-view');
  await loadGroup();
}
async function loadGroup(){
  if (!currentGroup) return;
  const v = await api('/api/groups/' + encodeURIComponent(currentGroup) + '/view');
  groupData = v;
  const run = v.chief;
  const key = v.messages.length + ':' + (run ? run.updated_at + run.status : '') + ':' + v.events_next;
  if (key === groupKey) return;
  groupKey = key;
  renderGroup();
}
function botOf(id){ return (groupData.bots || []).find(b => b.id === id); }
function stepFor(run, member){ return run ? run.steps.find(s => s.member === member) : null; }
function renderGroup(){
  const v = groupData, run = v.chief;
  $('group-title').textContent = v.name;
  $('group-sub').textContent = t('m.group.sub', { n: v.bots.length });
  const st = $('group-stack'); st.replaceWith(Object.assign(stack(v.bots.map(b => b.mascot)), { id: 'group-stack' }));
  // 进度
  const prog = $('group-prog'); prog.innerHTML = '';
  prog.classList.toggle('hidden', !run);
  if (run) {
    const tt = el('div', 't'); tt.appendChild(el('b', '', runProgress(run)));
    const segs = el('div', 'segs'); run.steps.forEach(s => segs.appendChild(el('i', s.status === 'done' ? 'ok' : s.status === 'running' ? 'run' : s.status === 'failed' ? 'bad' : ''))); tt.appendChild(segs);
    prog.appendChild(tt);
  }
  // @ 成员
  const at = $('group-at'); at.innerHTML = '';
  v.bots.forEach(b => { const x = el('button'); x.appendChild(mascot(b.mascot, 'idle', 20)); x.appendChild(document.createTextNode('@' + b.name)); x.onclick = () => { const i = $('group-input'); i.value = ('@' + b.id + ' ' + i.value.replace(/^@\S+\s*/, '')); i.focus(); }; at.appendChild(x); });
  const feed = $('group-feed'); feed.innerHTML = '';
  let overviewShown = false, summaryShown = false;
  v.messages.forEach(m => {
    if (m.sender === 'user') { feed.appendChild(el('div', 'u', m.content)); return; }
    if (m.sender === 'chief') {
      if (run && !overviewShown && /^分成|^Split into/.test(m.content)) { overviewShown = true; feed.appendChild(dispatchCard(run)); return; }
      if (run && /^@[a-z]/.test(m.content)) return; // 单步分派消息由分派卡表示（@成员 + 进度）
      if (run && run.summary && /^汇总：|^Summary:/.test(m.content)) { summaryShown = true; feed.appendChild(summaryCard(run)); return; }
      feed.appendChild(chiefMsg(m.content)); return;
    }
    const b = botOf(m.sender);
    const s = stepFor(run, m.sender);
    const gm = el('div', 'gm'); gm.appendChild(mascot(b ? b.mascot : 'nyra', m.error ? 'idle' : (s && s.status === 'done' ? 'done' : 'idle'), 28, { label: b ? b.name : m.sender }));
    const bd = el('div', 'bd'); const nm = el('div', 'nm', b ? b.name : m.sender);
    if (s) nm.appendChild(el('span', '', m.error ? t('m.group.failed') : t('m.group.returned')));
    bd.appendChild(nm); bd.appendChild(el('div', 'tx' + (m.error ? ' err' : ''), m.content)); gm.appendChild(bd); feed.appendChild(gm);
  });
  if (run && !overviewShown && run.status === 'running') feed.appendChild(chiefMsg(t('m.chief.starting'), 'think'));
  // 正在处理的成员
  const status = $('group-status'); status.innerHTML = '';
  const cur = run && run.steps.find(s => s.status === 'running');
  if (cur) { const b = botOf(cur.member); status.appendChild(mascot(b ? b.mascot : 'nyra', 'work', 20)); status.appendChild(document.createTextNode(t('m.group.working', { who: b ? b.name : cur.member }))); }
  window.scrollTo(0, document.body.scrollHeight);
}
function dispatchCard(run){
  const gm = el('div', 'gm'); gm.appendChild(mascot('chief', run.status === 'running' ? 'work' : 'idle', 28, { label: t('m.chief.name') }));
  const bd = el('div', 'bd'); const nm = el('div', 'nm', t('m.chief.name')); nm.appendChild(el('span', '', t('m.group.dispatch'))); bd.appendChild(nm);
  bd.appendChild(el('div', 'tx', t('m.group.dispatchLead', { n: run.steps.length })));
  const asg = el('div', 'asg'); asg.id = 'dispatch-card';
  run.steps.forEach(s => {
    const ar = el('div', 'ar'); ar.dataset.step = s.status;
    const key = memberMascot(run, s.member);
    ar.appendChild(mascot(key, s.status === 'running' ? 'work' : s.status === 'done' ? 'done' : 'idle', 26));
    const tt = el('div', 't'); const b = el('b'); b.appendChild(el('span', 'at', '@' + memberName(run, s.member))); b.appendChild(document.createTextNode(' ' + s.title)); tt.appendChild(b);
    if (s.excerpt && s.status !== 'pending') tt.appendChild(el('small', '', s.excerpt));
    const bar = el('div', 'bar'); const fill = el('i'); fill.style.width = s.status === 'done' || s.status === 'failed' ? '100%' : s.status === 'running' ? '50%' : '0';
    fill.style.background = s.status === 'done' ? 'var(--ok)' : s.status === 'failed' ? 'var(--bad)' : 'var(--brand)'; bar.appendChild(fill); tt.appendChild(bar);
    ar.appendChild(tt);
    ar.appendChild(el('span', 'st ' + (s.status === 'done' ? 'ok' : s.status === 'failed' ? 'bad' : s.status === 'running' ? 'run' : ''), t('m.step.' + s.status)));
    asg.appendChild(ar);
  });
  bd.appendChild(asg); gm.appendChild(bd); return gm;
}
function summaryCard(run){
  const gm = el('div', 'gm'); gm.appendChild(mascot('chief', run.status === 'done' ? 'done' : 'idle', 28, { label: t('m.chief.name') }));
  const bd = el('div', 'bd'); const nm = el('div', 'nm', t('m.chief.name')); nm.appendChild(el('span', '', t('m.group.summaryFor'))); bd.appendChild(nm);
  const sum = el('div', 'sum'); sum.id = 'summary-card'; sum.appendChild(el('div', 'h', run.summary.title));
  run.summary.lines.forEach(l => {
    const li = el('div', 'li'); li.appendChild(mascot(memberMascot(run, l.member), 'idle', 18));
    const sp = el('span'); const b = el('b', '', memberName(run, l.member)); b.style.fontWeight = '600'; sp.appendChild(b); sp.appendChild(document.createTextNode(t('m.colon') + l.text));
    if (!l.ok) sp.style.color = 'var(--bad-t)';
    li.appendChild(sp); sum.appendChild(li);
  });
  bd.appendChild(sum); gm.appendChild(bd); return gm;
}
async function sendGroup(){
  const i = $('group-input'); const text = i.value.trim(); if (!text || !currentGroup) return;
  i.value = ''; autoGrow(i);
  try { await api('/api/groups/' + encodeURIComponent(currentGroup) + '/messages', { method: 'POST', body: JSON.stringify({ text, async: true }) }); toast(t('m.group.sent')); }
  catch (e) { i.value = text; actionFailed(e, sendGroup); }
  groupKey = ''; await loadGroup().catch(() => {});
}

// ---------------------------------------------------------------- 字体 ---
// CSP 不放开 font-src：同源 fetch 拿到 woff 字节，再用 FontFace 从 ArrayBuffer 加载。
function loadFonts(){
  if (!window.FontFace || !document.fonts) return;
  [['Inter', 'inter-400', '400'], ['Inter', 'inter-600', '500 700'], ['JetBrains Mono', 'jbm-400', '400']].forEach(([fam, file, weight]) => {
    fetch('/m/fonts/' + file + '.woff').then(r => r.ok ? r.arrayBuffer() : Promise.reject())
      .then(buf => new FontFace(fam, buf, { weight }).load()).then(f => document.fonts.add(f)).catch(() => {});
  });
}

// ---------------------------------------------------------------- 入场动画 ---
function isStandalone(){
  try { return (window.matchMedia && matchMedia('(display-mode: standalone)').matches) || navigator.standalone === true; } catch(e){ return false; }
}
function isPhone(){
  const ua = navigator.userAgent || '';
  if (/Android|iPhone|iPod/.test(ua)) return true;
  try { return matchMedia('(pointer: coarse)').matches && Math.min(screen.width, screen.height) < 600; } catch(e){ return false; }
}
function shouldSplash(){
  if (!isStandalone() || !isPhone()) return false;
  try { if (sessionStorage.getItem('nyra-splash-shown')) return false; sessionStorage.setItem('nyra-splash-shown', '1'); } catch(e){}
  return true;
}
function runSplash(){
  const s = $('splash');
  if (!shouldSplash()) { s.remove(); return; }
  let reduce = false; try { reduce = matchMedia('(prefers-reduced-motion: reduce)').matches; } catch(e){}
  s.classList.remove('hidden'); s.dataset.state = 'playing';
  const finish = () => { s.dataset.state = 'done'; s.classList.add('out'); setTimeout(() => s.remove(), 480); };
  if (reduce) { s.classList.add('reduce'); s.innerHTML = splashSvg(SPLASH_END); setTimeout(finish, 1400); return; }
  s.classList.add('play');
  let start = null;
  const step = (ts) => {
    if (start === null) start = ts;
    const sec = (ts - start) / 1000;
    s.innerHTML = splashSvg(Math.min(sec, SPLASH_END));
    s.dataset.t = sec.toFixed(2);
    if (sec >= SPLASH_HOLD) { finish(); return; }
    if (sec >= SPLASH_END) { setTimeout(() => finish(), (SPLASH_HOLD - sec) * 1000); return; }
    requestAnimationFrame(step);
  };
  s.innerHTML = splashSvg(0);
  requestAnimationFrame(step);
}

// 页面滚动后顶部栏加一道细线（静止时与页面同色）。被动监听，只切一个 class。
(function(){
  let on = false;
  const sync = () => {
    const s = (window.scrollY || document.documentElement.scrollTop) > 4;
    if (s === on) return; on = s;
    document.documentElement.classList.toggle('scrolled', s);
    document.querySelectorAll('.cbar').forEach(h => h.classList.toggle('scrolled', s));
  };
  window.addEventListener('scroll', sync, { passive: true });
  sync();
})();
