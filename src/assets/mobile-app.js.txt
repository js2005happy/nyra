function deviceName(){
  const ua = navigator.userAgent || '';
  if (/iPhone/.test(ua)) return 'iPhone';
  if (/iPad/.test(ua)) return 'iPad';
  const m = /Android[^;]*;\s*([^;)]+?)(?:\sBuild|\))/.exec(ua);
  if (m) return m[1].trim().slice(0, 40);
  if (/Android/.test(ua)) return 'Android';
  return 'Phone';
}

function relTime(iso){
  if (!iso) return '';
  const d = new Date(/Z|[+-]\d\d:?\d\d$/.test(iso) ? iso : iso + 'Z');
  const s = Math.max(0, (Date.now() - d.getTime()) / 1000);
  if (!isFinite(s) || s < 60) return t('m.time.now');
  if (s < 3600) return t('m.time.m', { n: Math.floor(s / 60) });
  if (s < 86400) return t('m.time.h', { n: Math.floor(s / 3600) });
  return t('m.time.d', { n: Math.floor(s / 86400) });
}

function threadState(s){
  const a = lastApprovals.find(x => x.session === s.id);
  if (a) return ['wait', waitText(a)];
  if (s.status === 'running') return ['run', t('m.st.running')];
  if (s.status === 'error') return ['err', t('m.st.error')];
  const seen = seenMap()[s.id];
  const fresh = seen !== undefined && (s.count || 0) > seen;
  const o = s.outcome;
  if (o && o.steps) {
    const push = t(o.pushed ? 'm.sub.pushed' : 'm.sub.notPushed');
    const txt = o.edited && o.edited.length ? t(o.edited.length === 1 ? 'm.sub.doneEdit1' : 'm.sub.doneEdit', { n: o.edited.length, push }) : (o.commands ? '' : t('m.sub.doneRead'));
    if (txt) return [fresh ? 'review' : '', txt];
  }
  if (fresh) return ['review', t('m.st.review')];
  return ['', t('m.st.idle', { n: s.count || 0 })];
}

}

// 运行中的这一轮还没写进会话（网关只给出本轮之前的消息）：从事件流补上本轮的用户消息、工具步骤和中间回复。
function liveTurn(v){
  const out = [];
  if (v.status !== 'running' || !Array.isArray(v.events)) return out;
  let last = -1; v.events.forEach((e, i) => { if (e.type === 'user') last = i; });
  if (last < 0) return out;
  v.events.slice(last).forEach(e => {
    const d = e.data || {};
    if (e.type === 'user') out.push({ role: 'user', content: String(d.content || '') });
    else if (e.type === 'tool_start') out.push({ role: 'assistant', content: '', tool_calls: [{ function: { name: String(d.name || '?'), arguments: typeof d.arguments === 'string' ? d.arguments : JSON.stringify(d.arguments || {}) } }] });
    else if (e.type === 'assistant') out.push({ role: 'assistant', content: String(d.content || '') });
  });
  return out;
}
function renderFeed(v){
  const all = v.messages || [];
  // 正在等审批的那一步由审批卡表示，还没执行，不提前写进过程里（与桌面一致）。
  const live = liveTurn(v); if (lastApprovals.some(a => a.session === currentSession) && live.length && live[live.length - 1].tool_calls) live.pop();
  const msgs = all.filter(m => m.role !== 'system').concat(live);
  const key = msgs.length + ':' + (msgs.length ? textOf(msgs[msgs.length - 1]).length : 0) + ':' + v.status + ':' + (v.next || 0);
  $('chat-status').textContent = v.status === 'running' ? t('m.working') : (v.status === 'error' ? t('m.errorPrefix', { msg: v.error || '' }) : '');
  const firstUser = msgs.find(m => m.role === 'user');
  if (firstUser) $('chat-title').textContent = textOf(firstUser).slice(0, 60) || t('m.untitled');
  if (v.status !== 'running') markSeen(currentSession, all.length);
  renderCPills();
  if (key === lastFeedKey) return;
  lastFeedKey = key;
  const feed = $('feed'); feed.innerHTML = '';
  if (!msgs.length && v.status !== 'running' && !pendingText) { feed.appendChild(emptyState()); return; }
  let steps = [];
  const flush = (open) => { if (steps.length) { feed.appendChild(stepsBlock(steps, open)); steps = []; } };
  msgs.forEach(m => {
    if (m.role === 'tool') return;
    const txt = textOf(m);
    if (m.role === 'assistant' && m.tool_calls && m.tool_calls.length) {
      if (txt) { flush(false); feed.appendChild(bubble(m, all, txt)); }
      m.tool_calls.forEach(c => steps.push(stepOf(c)));
      return;
    }
    if (!txt) return;
    flush(false);
    feed.appendChild(bubble(m, all, txt));
  });
  flush(false);
  // 运行中的这一轮还没写进会话：先把刚发出的消息显示出来。
  if (pendingText && v.status === 'running' && !msgs.some(m => m.role === 'user' && textOf(m) === pendingText)) feed.appendChild(el('div', 'u', pendingText));
  if (v.status !== 'running') pendingText = '';
  window.scrollTo(0, document.body.scrollHeight);
}

function parseDiff(summary){
  const lines = String(summary || '').split('\n');
  let path = '', add = 0, del = 0; const body = [];
  lines.slice(1).forEach(l => {
    if (l.startsWith('+++ ')) { path = l.replace(/^\+\+\+ (b\/)?/, ''); return; }
    if (l.startsWith('--- ')) { if (!path) path = l.replace(/^--- (a\/)?/, ''); return; }
    if (l.startsWith('+')) { add++; body.push(['a', '+', l.slice(1)]); }
    else if (l.startsWith('-')) { del++; body.push(['d', '−', l.slice(1)]); }
    else if (l.trim()) body.push(['', '', l]);
  });
  return { path, add, del, body, head: lines[0] || '' };
}

function scheduleText(task){
  if (task.at) return t('m.auto.daily', { at: task.at });
  const s = Number(task.every_seconds || 0);
  if (s >= 3600 && s % 3600 === 0) return t('m.auto.everyHour', { n: s / 3600 });
  return t('m.auto.everyMin', { n: Math.round(s / 60) });
}
function fmtWhen(iso){
  try { const d = new Date(iso); return d.toLocaleString(LANG === 'zh' ? 'zh-CN' : 'en', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }); } catch(e){ return iso; }
}

    e.appendChild(el('code', '', t('m.review.emptyHint'))); box.appendChild(e); return;
  }
  const pending = items.filter(x => x.status === 'pending').length;
  box.appendChild(el('div', 'lh', t('m.review.pending') + ' ' + pending + ' / ' + items.length));
  items.forEach(item => {
    const card = el('div', 'rv');

    head.appendChild(el('span', 'tag', reviewStatusLabel(item)));

      card.appendChild(el('div', 'note warn-t', t('m.review.unverifiedWarn')));

  });
}

// CSP 不放开 font-src：同源 fetch 拿到 woff 字节，再用 FontFace 从 ArrayBuffer 加载。
function loadFonts(){
  if (!window.FontFace || !document.fonts) return;
  [['Inter', 'inter-400', '400'], ['Inter', 'inter-600', '500 700'], ['JetBrains Mono', 'jbm-400', '400']].forEach(([fam, file, weight]) => {
    fetch('/m/fonts/' + file + '.woff').then(r => r.ok ? r.arrayBuffer() : Promise.reject())
      .then(buf => new FontFace(fam, buf, { weight }).load()).then(f => document.fonts.add(f)).catch(() => {});
  });
}

function isStandalone(){
  try { return (window.matchMedia && matchMedia('(display-mode: standalone)').matches) || navigator.standalone === true; } catch(e){ return false; }
}
function isPhone(){
  const ua = navigator.userAgent || '';
  if (/Android|iPhone|iPod/.test(ua)) return true;
  try { return matchMedia('(pointer: coarse)').matches && Math.min(screen.width, screen.height) < 600; } catch(e){ return false; }
}
function shouldSplash(){
  if (!isStandalone() || !isPhone()) return false;
  try { if (sessionStorage.getItem('nyra-splash-shown')) return false; sessionStorage.setItem('nyra-splash-shown', '1'); } catch(e){}
  return true;
}
function setThemeColor(c){ document.querySelectorAll('meta[name=theme-color]').forEach(m => { if (!m.dataset.orig) m.dataset.orig = m.content; m.content = c || m.dataset.orig; }); }
function runSplash(){
  const s = $('splash');
  if (!shouldSplash()) { s.remove(); return; }
  let reduce = false; try { reduce = matchMedia('(prefers-reduced-motion: reduce)').matches; } catch(e){}
  s.classList.remove('hidden'); s.dataset.state = 'playing'; setThemeColor('#1a1a1a');
  const finish = () => {
    s.dataset.state = 'done'; s.classList.add('out'); setThemeColor('');
    setTimeout(() => s.remove(), 480);
  };
  if (reduce) { s.classList.add('reduce'); s.innerHTML = splashSvg(SPLASH_END); setTimeout(finish, 1400); return; }
  s.classList.add('play');
  let start = null;
  const step = (ts) => {
    if (start === null) start = ts;
    const sec = (ts - start) / 1000;
    s.innerHTML = splashSvg(Math.min(sec, SPLASH_END));
    s.dataset.t = sec.toFixed(2);
    if (sec >= SPLASH_HOLD) { finish(); return; }
    if (sec >= SPLASH_END) { setTimeout(() => finish(), (SPLASH_HOLD - sec) * 1000); return; }
    requestAnimationFrame(step);
  };
  s.innerHTML = splashSvg(0);
  requestAnimationFrame(step);
}

// 液态玻璃：页面滚动后顶部栏才变成玻璃（静止时与页面同色）。被动监听，只切一个 class，不逐帧改样式。
(function(){
  let on = false;
  const sync = () => {
    const s = (window.scrollY || document.documentElement.scrollTop) > 4;
    if (s === on) return; on = s;
    document.documentElement.classList.toggle('scrolled', s);
    const hd = document.querySelector('#chat-view .hd'); if (hd) hd.classList.toggle('scrolled', s);
  };
  window.addEventListener('scroll', sync, { passive: true });
  sync();
})();
