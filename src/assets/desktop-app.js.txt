function token(){
  const q = new URLSearchParams(location.search).get('token');
  if (q) {
    try { sessionStorage.setItem('nyra-token', q); } catch(e){}
    return q;
  }
  try { return sessionStorage.getItem('nyra-token') || ''; } catch(e){ return ''; }
}

function ago(iso){
  if (!iso) return '';
  const d = new Date(/[zZ]|[+-]\d\d:?\d\d$/.test(iso) ? iso : iso + 'Z');
  let s = (Date.now() - d.getTime()) / 1000;
  if (!isFinite(s)) return '';
  if (s < 0) { const d2 = new Date(iso); s = (Date.now() - d2.getTime()) / 1000; }
  if (s < 60) return t('d.ago.now');
  if (s < 3600) return t('d.ago.m', { n: Math.floor(s / 60) });
  if (s < 86400) return t('d.ago.h', { n: Math.floor(s / 3600) });
  return t('d.ago.d', { n: Math.floor(s / 86400) });
}

// 统一 diff（网关审批摘要里的 unifiedPreview）→ 文件名、增删数、逐行。网关预览不带行号，栏里显示 +/−。
function parseDiff(summary){
  const lines = String(summary || '').split('\n');
  let path = '', add = 0, del = 0; const body = [];
  lines.slice(1).forEach(l => {
    if (l.startsWith('+++ ')) { path = l.replace(/^\+\+\+ (b\/)?/, ''); return; }
    if (l.startsWith('--- ')) { if (!path) path = l.replace(/^--- (a\/)?/, ''); return; }
    if (l.startsWith('+')) { add++; body.push(['a', '+', l.slice(1)]); }
    else if (l.startsWith('-')) { del++; body.push(['d', '−', l.slice(1)]); }
    else if (l.trim()) body.push(['', '', l]);
  });
  return { path, add, del, body, head: lines[0] || '' };
}

function renderThreads(){
  const box = $('threads'); box.innerHTML = '';
  if (!S.sessions) return;
  const pend = new Set(S.approvals.map(a => a.session));
  const list = (S.sessions.sessions || []).slice().sort((a, b) => String(b.mtime).localeCompare(String(a.mtime)));
  const groups = new Map(); const chats = [];
  const here = projectName();
  if (here) groups.set(here, []);
  list.forEach(s => { if (s.project) { if (!groups.has(s.project)) groups.set(s.project, []); groups.get(s.project).push(s); } else chats.push(s); });
  const row = (s, chat) => {
    const b = el('button', 'th' + (chat ? ' chat' : '') + (s.id === S.session ? ' on' : ''));
    b.dataset.sid = s.id;
    if (s.status === 'running' && !pend.has(s.id)) b.appendChild(el('span', 'spin'));
    else if (pend.has(s.id)) b.appendChild(el('span', 'dot warn'));
    else if (s.status === 'error') b.appendChild(el('span', 'dot err'));
    b.appendChild(el('span', 'tt', titleOf(s) || t('d.untitled')));
    b.appendChild(el('span', 't', s.status === 'running' ? t('d.ago.now') : ago(s.mtime)));
    b.title = titleOf(s);
    b.onclick = () => openSession(s.id);
    return b;
  };
  groups.forEach((items, name) => {
    const closed = !!S.collapsed[name];
    const h = el('button', 'pr'); h.appendChild(el('span', 'car', closed ? '▸' : '▾')); h.appendChild(document.createTextNode(name));
    h.onclick = () => { S.collapsed[name] = !closed; renderThreads(); };

    if (!items.length) box.appendChild(el('div', 'side-empty', t('d.noThreads')));
    items.slice(0, 30).forEach(s => box.appendChild(row(s, false)));
  });
  if (chats.length) {
    const sh = el('div', 'sh'); sh.appendChild(el('span', '', t('d.chats'))); box.appendChild(sh);
    chats.slice(0, 30).forEach(s => box.appendChild(row(s, true)));
  }
}

function scheduleWhen(task){
  if (task.enabled === false) return t('m.auto.paused');
  if (task.at) return t('m.auto.daily', { at: task.at });
  const sec = Number(task.every_seconds || 0);
  if (sec >= 3600 && sec % 3600 === 0) return t('m.auto.everyHour', { n: sec / 3600 });
  if (sec) return t('m.auto.everyMin', { n: Math.round(sec / 60) });
  return '';
}

function stepRow(name, args){
  const r = el('div', 'r');
  const put = (ic, verb, code) => { r.appendChild(icon(ic)); r.appendChild(document.createTextNode(t(verb))); if (code) r.appendChild(el('span', 'c', code)); };
  if (name === 'shell' && args.command) put('play', 'm.step.ran', String(args.command));
  else if (name === 'read_file' && args.path) put('diamond', 'm.step.read', String(args.path));
  else if (EDIT_TOOLS.includes(name) && args.path) put('pencil', 'm.step.edited', String(args.path));
  else if (/search|grep|glob|list/.test(name)) put('search', 'm.step.called', name + (args.pattern || args.query ? ' "' + (args.pattern || args.query) + '"' : ''));
  else put('dot', 'm.step.called', name + (args.path ? ' ' + args.path : ''));
  return r;
}
function argsOf(c){ let a = {}; try { a = typeof c.function.arguments === 'string' ? JSON.parse(c.function.arguments || '{}') : (c.function.arguments || {}); } catch(e){} return a; }
// 运行中的这一轮还没写进会话：从事件流补上本轮的用户消息、工具步骤和中间回复（与手机端一致）。
function liveTurn(v){
  const out = [];
  if (v.status !== 'running' || !Array.isArray(v.events)) return out;
  let last = -1; v.events.forEach((e, i) => { if (e.type === 'user') last = i; });
  if (last < 0) return out;
  v.events.slice(last).forEach(e => {
    const d = e.data || {};
    if (e.type === 'user') out.push({ role: 'user', content: String(d.content || '') });
    else if (e.type === 'tool_start') out.push({ role: 'assistant', content: '', tool_calls: [{ function: { name: String(d.name || '?'), arguments: typeof d.arguments === 'string' ? d.arguments : JSON.stringify(d.arguments || {}) } }] });
    else if (e.type === 'assistant') out.push({ role: 'assistant', content: String(d.content || '') });
  });
  return out;
}
// 本会话里审批过的编辑：路径 → 统一 diff（改动页与“改动文件”卡片的 +/− 用它，不编数字）。
function diffsByPath(v){
  const out = {};
  // 被拒绝的编辑没发生，不算进改动。
  const denied = new Set((v.events || []).filter(e => e.type === 'approval_result' && e.data && e.data.approved === false && e.data.id).map(e => e.data.id));
  (v.events || []).forEach(e => { const d = e.data || {}; if (e.type === 'approval' && EDIT_TOOLS.includes(d.tool) && d.summary && !denied.has(d.id)) { const p = parseDiff(d.summary); if (p.path) out[p.path] = p; } });
  S.approvals.filter(a => a.session === S.session && EDIT_TOOLS.includes(a.tool)).forEach(a => { const p = parseDiff(a.summary); if (p.path) out[p.path] = p; });
  return out;
}

function avatar(state){ const tpl = $('av-' + state); const s = el('span', 'av'); if (tpl) s.appendChild(tpl.content.firstElementChild.cloneNode(true)); return s; }
function fmtDur(s){ s = Math.max(1, Math.round(s)); return s < 60 ? t('d.dur.s', { s }) : t('d.dur.ms', { m: Math.floor(s / 60), s: s % 60 }); }
function isPush(cmd){ return /(^|[;&|(\s])git(\s+-[^\s]+(\s+[^\s-][^\s]*)?)*\s+push(\s|$)/.test(String(cmd || '')); }
function cap(s){ return s ? s.charAt(0).toUpperCase() + s.slice(1) : s; }
// 只数真正做成了的：编辑看工具结果（“已写入 / 已编辑 / 已应用补丁”），命令看有结果且没被拒绝，读看有结果且没报错。
// 被拒绝、报错、还没出结果的调用只算进“显示过程（N 步）”，不算进收尾交代。
function summarize(list){
  const res = {};
  list.forEach(m => { if (m.role === 'tool' && m.tool_call_id) res[m.tool_call_id] = textOf(m); });
  const edited = [], read = new Set(); let steps = 0, cmds = 0, pushed = false;
  list.forEach(m => {
    if (m.role !== 'assistant' || !m.tool_calls) return;
    m.tool_calls.forEach(c => {
      steps++; const n = (c.function && c.function.name) || ''; const a = argsOf(c);
      const r = c.id != null ? res[c.id] : undefined;
      if (r === undefined) return;
      if (EDIT_TOOLS.includes(n) && a.path) { if (/^已(写入|编辑|应用补丁)/.test(r) && !edited.includes(String(a.path))) edited.push(String(a.path)); }
      else if (n === 'read_file' && a.path) { if (!/^错误：/.test(r)) read.add(String(a.path)); }
      else if (n === 'shell' && a.command) { if (!/^错误：用户拒绝/.test(r)) { cmds++; if (isPush(a.command)) pushed = true; } }
    });
  });
  return { steps, edited, read: read.size, cmds, pushed };
}
// 收尾交代：只说工具调用里看得到的事实。跑过 shell 却没用编辑工具时不说“没改文件”（命令可能改了文件）。
function wrapText(sm){
  const cmds = sm.cmds ? t(sm.cmds === 1 ? 'd.wrap.cmds1' : 'd.wrap.cmds', { n: sm.cmds }) : '';
  let s = '';
  if (sm.edited.length) {
    s = t(sm.edited.length === 1 ? 'd.wrap.edited1' : 'd.wrap.edited', { n: sm.edited.length });
    if (cmds) s += t('d.wrap.and') + cmds;
    s += t('d.wrap.sep') + t(sm.pushed ? 'd.wrap.pushed' : 'd.wrap.notPushed');
    const sb = S.info && S.info.config ? S.info.config.sandbox : 'local';
    if (sb !== 'e2b') s += t('d.wrap.sep') + t('d.wrap.checkpoint');
  } else {
    const bits = [];
    if (sm.read) bits.push(t(sm.read === 1 ? 'd.wrap.read1' : 'd.wrap.read', { n: sm.read }));
    if (cmds) bits.push(cmds);
    if (!bits.length) return '';
    s = bits.join(t('d.wrap.and'));
    if (sm.pushed) s += t('d.wrap.sep') + t('d.wrap.pushed');
    else if (!sm.cmds) s += t('d.wrap.sep') + t('d.wrap.noEdit');
  }
  return cap(s) + t('d.wrap.end');
}
// 事件流 → 每轮的开始/结束时间、token 增量、正在跑的工具。回退（rewind）之后重新计。
function eventTurns(v){
  const turns = []; let cur = null;
  (v.events || []).forEach(e => {
    const d = e.data || {};
    if (e.type === 'rewind') { turns.length = 0; cur = null; }
    if (e.type === 'user') {
      // 起点：网关在 user 事件里记的累计用量（事件流每轮清空，不能拿上一轮的数当起点）。没有起点就不算 token。
      const ub = d.usage_base && typeof d.usage_base === 'object' ? d.usage_base : null;
      const base = ub && ub.total_tokens != null ? Number(ub.total_tokens) : null;
      const baseMiss = ub ? Number(ub.missing_calls || 0) : 0;
      cur = { content: String(d.content || ''), start: e.ts, end: e.ts, base, baseMiss, tot: null, miss: baseMiss, tool: null }; turns.push(cur); return;
    }
    if (cur) cur.end = e.ts;
    if (e.type === 'usage') {
      const tt = d.total_tokens != null ? Number(d.total_tokens) : (d.prompt_tokens != null || d.completion_tokens != null ? Number(d.prompt_tokens || 0) + Number(d.completion_tokens || 0) : null);
      if (tt != null && isFinite(tt) && cur) { cur.tot = tt; cur.miss = Number(d.missing_calls || 0); }
    }
    if (cur) { if (e.type === 'tool_start') cur.tool = { name: String(d.name || ''), args: d.arguments }; else if (e.type === 'tool_result' || e.type === 'assistant') cur.tool = null; }
  });
  // 有调用没报用量（missing_calls 变多）时 token 不完整，不显示。
  turns.forEach(x => { x.tokens = (x.tot != null && x.base != null && x.miss === x.baseMiss) ? x.tot - x.base : null; if (!(x.tokens > 0)) x.tokens = null; });
  return turns;
}
function liveStepText(tool){
  if (!tool || !tool.name) return t('d.live.thinking');
  let a = tool.args; if (typeof a === 'string') { try { a = JSON.parse(a); } catch(e){ a = {}; } } a = a || {};
  if (EDIT_TOOLS.includes(tool.name) && a.path) return t('d.live.writing', { path: String(a.path) });
  if (tool.name === 'shell' && a.command) return t('d.live.running', { cmd: String(a.command).split('\n')[0].slice(0, 80) });
  if (tool.name === 'read_file' && a.path) return t('d.live.reading', { path: String(a.path) });
  return t('d.live.calling', { tool: tool.name });
}

function renderFeed(v){
  const all = v.messages || [];
  const pend = S.approvals.find(a => a.session === S.session) || null;
  // 正在等审批的那一步由审批卡表示，还没执行，不在步骤里提前写成“运行 / 编辑”
  const live = liveTurn(v); if (pend && live.length && live[live.length - 1].tool_calls) live.pop();
  const msgs = all.filter(m => m.role !== 'system').concat(live);
  const card = pend ? S.card[pend.id] || {} : {};
  const evTurns = eventTurns(v);
  const curTool = v.status === 'running' && evTurns.length ? (evTurns[evTurns.length - 1].tool || {}).name || '' : '';
  const key = [msgs.length, msgs.length ? textOf(msgs[msgs.length - 1]).length : 0, v.status, v.next || 0, pend ? pend.id + (card.open || '') + (card.scope || '') + ':' + Math.floor(Date.now() / 10000) : '', S.pending, S.review.open ? 1 : 0, curTool, (S.info && S.info.mode) || ''].join(':');
  const fu = msgs.find(m => m.role === 'user'); const newTitle = fu ? textOf(fu) : (S.pending || '');
  if (newTitle && newTitle !== S.title) { S.title = newTitle; renderTop(); renderThreads(); }
  if (key === S.feedKey) { tickLive(); return; }
  const feedEl = $('feed');
  const nearBottom = feedEl.scrollHeight - feedEl.scrollTop - feedEl.clientHeight < 120;
  const keepFeedback = document.querySelector('#col .appr textarea, #rbody .appr-more textarea');
  const fbText = keepFeedback ? keepFeedback.value : null;
  S.feedKey = key;
  const col = $('col'); col.innerHTML = '';
  // 按轮切：用户消息 → Nyra（过程收起成“显示过程（N 步）”）→ 回复 → 收尾交代 → 用时 / token。
  const turns = []; let cur = null;
  msgs.forEach(m => {
    if (m.role === 'user') { cur = { user: m, items: [] }; turns.push(cur); return; }
    if (!cur) { cur = { user: null, items: [] }; turns.push(cur); }
    cur.items.push(m);
  });
  // 事件里的轮次与消息里的用户消息从后往前对齐；内容开头对不上就不用（不张冠李戴）。
  const userTurns = turns.filter(x => x.user);
  const k = Math.min(evTurns.length, userTurns.length);
  for (let j = 0; j < k; j++) {
    const tu = userTurns[userTurns.length - k + j], et = evTurns[evTurns.length - k + j];
    const a = textOf(tu.user).trim().slice(0, 40), b = et.content.trim().slice(0, 40);
    if (a && b && (a.startsWith(b) || b.startsWith(a))) tu.ev = et;
  }
  const running = v.status === 'running';
  const extra = !!(S.pending && running && !msgs.some(m => m.role === 'user' && textOf(m) === S.pending));
  turns.forEach((tu, ti) => {
    const last = ti === turns.length - 1 && !extra;
    const done = !(last && (running || pend));
    if (tu.ev) tu.ev.done = done;
    if (tu.user) col.appendChild(bubble(tu.user, all, textOf(tu.user), 'u'));
    // 最终回复 = 本轮最后一条没有工具调用、有文字的 assistant 消息（本轮结束后才算）。
    let finalIdx = -1;
    if (done) for (let i = tu.items.length - 1; i >= 0; i--) { const m = tu.items[i]; if (m.role === 'assistant' && !(m.tool_calls && m.tool_calls.length) && textOf(m)) { finalIdx = i; break; } }
    const proc = []; let n = 0;
    tu.items.forEach((m, i) => {
      if (i === finalIdx || m.role === 'tool') return;
      const txt = textOf(m);
      if (txt) proc.push(el('div', 'r txt', txt));
      (m.tool_calls || []).forEach(c => { n++; proc.push(stepRow((c.function && c.function.name) || '?', argsOf(c))); });
    });
    const hasNyra = tu.items.some(m => m.role === 'assistant') || (last && (running || pend));
    if (!hasNyra) return;
    const liveNow = last && running && !pend;
    // 工作中：只有一行实时状态（团子 · Nyra · 当前步骤 · 用时），已做的步骤收在它下面。
    if (liveNow) col.appendChild(liveLine(tu.ev || null));
    else { const who = el('div', 'who-row'); who.appendChild(avatar(last && pend ? 'wait-s' : 'done-s')); who.appendChild(el('span', '', 'Nyra')); col.appendChild(who); }
    if (n) col.appendChild(processBlock(proc, n, false));
    else proc.forEach(p => { p.className = 'p'; col.appendChild(p); });
    if (finalIdx >= 0) {
      const m = tu.items[finalIdx]; const txt = textOf(m);
      const b = bubble(m, all, txt, 'p'); col.appendChild(b);
      const sm = summarize(tu.items);
      const w = sm.steps ? wrapText(sm) : '';
      if (w) { const wl = el('div', 'wrapup'); wl.appendChild(icon('check')); wl.appendChild(el('span', '', w)); col.appendChild(wl); }
      col.appendChild(replyFoot(txt, tu.ev));
      if (sm.edited.length) {
        const diffs = diffsByPath(v); const f = el('div', 'files');
        sm.edited.forEach(p => {
          const d = diffs[p]; const r = el('button', 'fr'); r.appendChild(el('span', 'mono', p)); r.firstChild.style.fontSize = '12px';
          if (d) { const a = el('span', 'add', '+' + d.add); a.style.marginLeft = 'auto'; r.appendChild(a); r.appendChild(el('span', 'del', '−' + d.del)); }
          r.onclick = () => { S.review.open = true; S.review.tab = 'diff'; S.review.file = p; S.feedKey = ''; renderTop(); renderReview(); tick(); };
          f.appendChild(r);
        });
        col.appendChild(f);
      }
    }
    if (last && pend) col.appendChild(approvalCard(pend, fbText, !!S.review.open));
  });
  if (extra) { col.appendChild(bubble({ role: 'user' }, all, S.pending, 'u')); col.appendChild(liveLine(null)); }
  if (running && !turns.length && !S.pending) col.appendChild(liveLine(null));
  if (!running) S.pending = '';
  if (v.status === 'error' && v.error) col.appendChild(el('div', 'err-line', t('d.errorPrefix', { msg: v.error })));
  // P0-3：审批超时自动拒绝与被打断的轮次，在对话里留一条可见记录（只看本轮事件流，不编造）。
  (v.events || []).forEach(e => { if (e.type === 'approval_result' && e.data && e.data.timeout) col.appendChild(el('div', 'note-line', t('d.appr.timeoutNote', { tool: String(e.data.tool || '') }))); });
  if (v.interrupted) col.appendChild(el('div', 'note-line', t('d.stoppedNote')));
  if (!msgs.length && !running) col.appendChild(el('div', 'empty', t('d.threadEmpty')));
  if (nearBottom || pend) feedEl.scrollTop = feedEl.scrollHeight;
}

  [['m.rewind.truncate', 'truncate'], ['m.rewind.edit', 'edit'], ['m.rewind.fork', 'fork']].forEach(([label, action]) => {

async function doRewind(index, action, text){
  if (!S.session) return;
  const rows = await api('/api/sessions/' + encodeURIComponent(S.session) + '/history').catch(() => null);
  const messageId = rows && rows.messages && rows.messages[index] ? rows.messages[index].id : null;
  if (messageId === null) { toast(t('m.rewind.notYet')); return; }
  let edited = null;
  if (action === 'edit') { edited = prompt(t('m.rewind.editPrompt'), text || ''); if (edited === null) return; }
  try {
    const res = await post('/api/sessions/' + encodeURIComponent(S.session) + '/rewind', { message_id: messageId, action, text: edited });
    if (res && res.new_session_id) { await loadSessions(); await openSession(res.new_session_id); return; }
    S.feedKey = ''; await tick();
  } catch (e) { toast(t('m.rewind.failed', { msg: e.message })); }
}

    toast(t('d.undone')); S.feedKey = ''; await tick();

}

function apprInfo(a){
  const isEdit = EDIT_TOOLS.includes(a.tool);
  const isShell = a.tool === 'shell' && !!a.command;
  const d = isEdit ? parseDiff(a.summary) : null;
  const base = (p) => String(p || '').split(/[\\/]/).pop();
  let sentence, compact;
  const rows = [];

    // 审批卡要盖住真正发出的参数：摘要全文按行展开（邮件正文/抄送/密送、
    // POST 请求体、键入全文、MCP 参数都在摘要里），不再只给首行前 200 字。

function shellRuns(v){
  const out = []; let cur = null;
  (v.events || []).forEach(e => {
    const d = e.data || {};
    if (e.type === 'tool_start' && d.name === 'shell') { const a = typeof d.arguments === 'string' ? (() => { try { return JSON.parse(d.arguments); } catch(x){ return {}; } })() : (d.arguments || {}); cur = { cmd: String(a.command || ''), out: null }; out.push(cur); }
    else if (e.type === 'tool_result' && d.name === 'shell' && cur) { cur.out = String(d.result || ''); cur = null; }
  });
  return out;
}

  // 触顶：预留失败文案里只有月度层被绑住时才带“月度支出上限”这句，按它认，不猜。
  if (v && v.status === 'error' && typeof v.error === 'string' && v.error.includes('月度支出上限')) { const c = add('', t('d.budgetHit'), 'warn'); c.style.cursor = 'pointer'; c.title = t('d.budgetHit'); c.onclick = () => openSettings('general'); }

  const right = el('span', 'r'); right.appendChild(el('span', 'dot ' + (n ? 'ok' : 'off'))); right.appendChild(document.createTextNode(n ? t('d.mcpOn', { n }) : t('d.mcpOff'))); st.appendChild(right);
}
function fmtK(n){ return n >= 1000 ? (n / 1000).toFixed(n >= 10000 ? 1 : 2).replace(/\.?0+$/, '') + 'k' : String(n); }
function fmtUsd(n){ return '$' + Number(n).toFixed(4).replace(/\.?0+$/, ''); }

// ---------------------------------------------------------------- 轮询 ---
let ticking = false;
async function tick(){
  if (ticking) return; ticking = true;
  try {
    await loadApprovals();
    if (S.view === 'thread' && S.session) {
      const sid = S.session;
      const v = await api('/api/sessions/' + encodeURIComponent(sid));
      if (sid === S.session) { S.sv = v; renderFeed(v); renderReview(); renderStatus(); }
      // P0-3：运行状态变了才重画输入行（停止/撤销按钮跟着状态走），平时不动，避免每秒重建 DOM。
      const pk = (S.sv && S.sv.status === 'running' ? 'run' : 'idle') + ':' + (S.sv && S.sv.messages && S.sv.messages.length ? 'msg' : 'nomsg');
      if (pk !== S.pillsKey) { S.pillsKey = pk; renderPills(); }
    }
    renderThreads();
  } catch (e) { /* 网关重启或断网：下一轮再试 */ }
  finally { ticking = false; }
}

  const inp = el('input', 'fld'); inp.type = 'password'; inp.autocomplete = 'off'; inp.spellcheck = false; inp.placeholder = t('d.key.placeholder'); inp.dataset.role = 'key-input';

  const res = el('div', 'result'); res.style.width = '100%';
  test.onclick = async () => {
    res.className = 'result'; res.textContent = t('d.key.testing'); test.disabled = true;

    catch (e) { res.className = 'result bad-t'; res.textContent = e.message; }
    test.disabled = false;
  };
  save.onclick = async () => {
    const v = inp.value.trim(); if (!v) { inp.focus(); return; }

  setTimeout(() => inp.focus(), 0);
  return box;
}

async function maybeSetup(force){
  let s; try { s = await api('/api/setup'); } catch (e) { return; }
  if (!s.needs_setup && !force) return;
  S.setupData = s;
  S.setup.provider = S.setup.provider || (s.providers.find(p => p.provider === s.provider && (p.needs_key || p.local)) || {}).provider || 'deepseek';
  S.setup.tested = null;
  $('setup').classList.remove('hidden');
  renderSetup();
}
const BRAND = { deepseek: 'DeepSeek', openai: 'OpenAI', gemini: 'Gemini', qwen: 'Qwen', kimi: 'Kimi', glm: 'GLM', mistral: 'Mistral', groq: 'Groq', together: 'Together', xai: 'xAI', openrouter: 'OpenRouter', ollama: 'Ollama', codex: 'Codex' };
const brand = (id) => BRAND[id] || id;
const SETUP_ORDER = ['deepseek', 'openai', 'gemini', 'qwen', 'kimi', 'glm', 'mistral', 'groq', 'together', 'xai', 'openrouter', 'ollama'];
function renderSetup(){
  const s = S.setupData; const box = $('setup-provs'); box.innerHTML = '';
  const provs = s.providers.filter(p => p.needs_key || p.provider === 'ollama').filter(p => p.provider !== 'custom');
  provs.sort((a, b) => (SETUP_ORDER.indexOf(a.provider) + 1 || 99) - (SETUP_ORDER.indexOf(b.provider) + 1 || 99));
  provs.forEach(p => {
    const b = el('button', 'prov' + (p.provider === S.setup.provider ? ' on' : '')); b.dataset.provider = p.provider;
    b.appendChild(el('b', '', brand(p.provider))); b.appendChild(el('small', '', p.local ? t('d.setup.noKey') : (p.source === 'missing' ? (p.default_model || '') : sourceLabel(p))));
    b.onclick = () => { S.setup.provider = p.provider; S.setup.tested = null; $('setup-key').value = ''; renderSetup(); };
    box.appendChild(b);
  });
  const cur = provs.find(p => p.provider === S.setup.provider) || provs[0];
  const local = cur && cur.local;
  $('setup-step2').classList.toggle('hidden', !!local);
  $('setup-key').placeholder = cur && cur.env_key ? t('d.setup.keyPh', { env: cur.env_key }) : '';
  $('setup-store').textContent = s.vault.unlocked ? t('d.key.storeVault') : t('d.key.storeFile');
  const res = $('setup-result');
  if (!S.setup.tested) { res.className = 'result'; res.textContent = local ? t('d.setup.localHint') : ''; }
  else showTest(res, S.setup.tested);
  $('setup-save').disabled = !(S.setup.tested && S.setup.tested.ok) || S.setup.busy;
}
async function setupTest(){
  const key = $('setup-key').value.trim();
  const cur = S.setupData.providers.find(p => p.provider === S.setup.provider);
  if (cur && !cur.local && !key) { $('setup-key').focus(); return; }
  const res = $('setup-result'); res.className = 'result'; res.textContent = t('d.key.testing');
  $('setup-test').disabled = true;
  try { S.setup.tested = await post('/api/keys/test', { provider: S.setup.provider, key: key || undefined }); }
  catch (e) { S.setup.tested = { ok: false, error: e.message }; }
  $('setup-test').disabled = false;
  renderSetup();
}
async function setupSave(){
  const key = $('setup-key').value.trim();
  const cur = S.setupData.providers.find(p => p.provider === S.setup.provider);
  S.setup.busy = true; renderSetup();
  try {
    if (cur && !cur.local) { const r = await post('/api/keys', { provider: S.setup.provider, key }); toast(t(r.stored === 'vault' ? 'd.key.savedVault' : 'd.key.savedFile', { mask: r.masked })); }
    $('setup-key').value = '';
    await post('/api/settings', { default_provider: S.setup.provider });
    await loadInfo();
    $('setup').classList.add('hidden');
    renderPills(); if (S.view === 'home') renderHome();
    $('home-input').focus();
  } catch (e) { const res = $('setup-result'); res.className = 'result bad-t'; res.textContent = e.message; }
  S.setup.busy = false; if (!$('setup').classList.contains('hidden')) renderSetup();
}

// ---------------------------------------------------------------- 入场动画 ---
// 只在桌面外壳（/api/info 的 shell === "desktop"）里、每次冷启动播一次（sessionStorage 随窗口进程清空）；
// 不能点击跳过；减少动态效果时只淡入终帧。
function playSplash(){
  return new Promise(resolve => {
    let played = false; try { played = sessionStorage.getItem('nyra-splash') === '1'; } catch(e){}
    if (played || typeof NyraSplash === 'undefined') { resolve(); return; }
    try { sessionStorage.setItem('nyra-splash', '1'); } catch(e){}
    const sp = $('splash'); sp.classList.remove('hidden');
    const svgAt = (tm) => NyraSplash.frame(tm).replace(/width="\d+" height="\d+"/, 'width="100%" height="100%" preserveAspectRatio="xMidYMid meet"');
    const end = () => { sp.classList.add('fade'); setTimeout(() => { sp.classList.add('hidden'); sp.innerHTML = ''; resolve(); }, 460); };
    if (matchMedia('(prefers-reduced-motion: reduce)').matches) {
      sp.innerHTML = svgAt(SPLASH_END); sp.classList.add('fadein');
      setTimeout(end, 1400); return;
    }
    const t0 = performance.now();
    const step = (now) => {
      const tm = Math.min((now - t0) / 1000, SPLASH_END);
      sp.innerHTML = svgAt(tm);
      if ((now - t0) / 1000 < SPLASH_HOLD) requestAnimationFrame(step); else end();
    };
    requestAnimationFrame(step);
  });
}

// ---------------------------------------------------------------- 字体 ---
function loadFonts(){
  if (!window.FontFace || !document.fonts) return;
  [['Inter', 'inter-400', '400'], ['Inter', 'inter-600', '500 600'], ['JetBrains Mono', 'jbm-400', '400']].forEach(([fam, file, weight]) => {
    fetch('/m/fonts/' + file + '.woff').then(r => r.ok ? r.arrayBuffer() : Promise.reject())
      .then(buf => new FontFace(fam, buf, { weight }).load()).then(f => document.fonts.add(f)).catch(() => {});
  });
}

function bind(){
  $('nav-new').onclick = newThread;
  $('nav-search').onclick = openSearch;
  $('nav-automations').onclick = openAutomations;
  $('nav-skills').onclick = openSkills;
  $('nav-mcp').onclick = openMcp;
  $('nav-settings').onclick = () => openSettings();
  $('kbd-new').textContent = IS_MAC ? '⌘N' : 'Ctrl N';
  $('kbd-search').textContent = IS_MAC ? '⌘K' : 'Ctrl K';
  ['home-input', 'input'].forEach(id => {
    const ta = $(id);
    ta.addEventListener('input', () => autoGrow(ta));
    ta.addEventListener('keydown', (e) => { if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) { e.preventDefault(); id === 'input' ? sendMsg() : sendHome(); } });
  });
  $('search-q').addEventListener('input', renderSearch);
  $('search-q').addEventListener('keydown', (e) => { if (e.key === 'Enter') { const b = $('search-res').querySelector('button'); if (b) b.click(); } });
  $('search-dlg').addEventListener('click', (e) => { if (e.target === $('search-dlg')) closeSearch(); });
  $('setup-test').onclick = setupTest;
  $('setup-save').onclick = setupSave;
  $('setup-key').addEventListener('input', () => { S.setup.tested = null; renderSetup(); });
  $('setup-key').addEventListener('keydown', (e) => { if (e.key === 'Enter') { e.preventDefault(); S.setup.tested && S.setup.tested.ok ? setupSave() : setupTest(); } });
  $('setup-skip').onclick = () => { $('setup-key').value = ''; $('setup').classList.add('hidden'); };
  document.addEventListener('keydown', (e) => {
    const mod = IS_MAC ? e.metaKey : e.ctrlKey;
    if (mod && (e.key === 'n' || e.key === 'N')) { e.preventDefault(); newThread(); }
    else if (mod && (e.key === 'k' || e.key === 'K')) { e.preventDefault(); openSearch(); }
    else if (e.key === 'Escape') { closeMenus(); closeSearch(); }
  });
}
async function boot(){
  applyI18n(); bind(); loadFonts();
  if (!TOKEN) { document.body.innerHTML = ''; document.body.appendChild(el('p', 'empty', t('d.noToken'))); return; }
  let info = null; try { info = await loadInfo(); } catch (e) { /* 下面照常渲染，轮询会重试 */ }
  // 设置里保存的语言优先于系统语言（P0-3：设置改语言要真换文案）；没保存才跟随系统。
  if (info && (info.language === 'zh' || info.language === 'en') && info.language !== LANG) { applyLang(info.language); applyI18n(); }
  const splash = info && info.shell === 'desktop' ? playSplash() : Promise.resolve();
  try { await loadSessions(); } catch (e) {}
  await loadSchedule();
  show('home'); renderHome(); renderPills();
  await splash;
  await maybeSetup(false);
  setInterval(tick, 1000);
  setInterval(() => { loadSessions().catch(() => {}); }, 4000);
  setInterval(() => { loadInfo().catch(() => {}); }, 15000);
}
boot();
